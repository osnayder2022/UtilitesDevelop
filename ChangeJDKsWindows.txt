Script para powershell de windows:

function Set-Java8 {
    $env:JAVA_HOME = "C:\Program Files\Java\jdk8u412"
    $env:Path = "$env:JAVA_HOME\bin;" + $env:Path
    Write-Host "Cambiado a Java 8" -ForegroundColor Green
}

function Set-Java21 {
    $env:JAVA_HOME = "C:\Program Files\Java\jdk21liberica"
    $env:Path = "$env:JAVA_HOME\bin;" + $env:Path
    Write-Host "Cambiado a Java 21" -ForegroundColor Green
}
